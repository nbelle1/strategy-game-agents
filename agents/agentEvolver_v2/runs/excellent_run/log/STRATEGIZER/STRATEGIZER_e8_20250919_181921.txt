```python
def _edge_to_nearest_open_info(self, game, start_edge):
    """Return (best_distance, best_pip_sum) for nearest open node reachable along graph from start_edge.
       If no open node within ROAD_MAX_DISTANCE, return (None, 0.0)."""
    try:
        board = game.state.board
        # get nodes touching the start edge
        start_nodes = board.edge_nodes(start_edge)
    except Exception:
        # fallback: no info
        return (None, 0.0)

    from collections import deque
    visited_edges = set()
    visited_nodes = set()
    q = deque()
    # push neighbor nodes with distance 0
    for n in start_nodes:
        q.append((n, 0))
        visited_nodes.add(n)

    best_dist = None
    best_pip = 0.0

    while q:
        node, dist = q.popleft()
        if dist > ROAD_MAX_DISTANCE:
            continue
        # check owner
        try:
            owner = board.node_owner(node)
        except Exception:
            try:
                owner = board.nodes[node].owner
            except Exception:
                owner = None
        if owner is None:
            # compute pip sum of this node's tiles
            pip_sum = 0.0
            try:
                tiles = board.node_tiles(node)
                for t in tiles:
                    # try tile.pip probability first, else fallback to pip attribute
                    pip = getattr(t, 'pip_probability', None)
                    if pip is None:
                        pip = getattr(t, 'pip', None)
                    if pip is None:
                        try:
                            pip = board.tile_pip_probability(t)
                        except Exception:
                            pip = 1.0
                    pip_sum += float(pip)
            except Exception:
                pip_sum = 0.0

            # prefer closer nodes with higher pip_sum
            if best_dist is None or dist < best_dist or (dist == best_dist and pip_sum > best_pip):
                best_dist = dist
                best_pip = pip_sum
            # do not early return; maybe there's closer better node
        # expand neighbors: get incident edges then adjacent nodes
        try:
            incident_edges = board.node_edges(node)
            for e in incident_edges:
                if e in visited_edges:
                    continue
                visited_edges.add(e)
                # get other node(s) for this edge
                try:
                    neigh_nodes = board.edge_nodes(e)
                except Exception:
                    continue
                for nn in neigh_nodes:
                    if nn in visited_nodes:
                        continue
                    visited_nodes.add(nn)
                    q.append((nn, dist + 1))
        except Exception:
            pass

    return (best_dist, best_pip)
```

        q = deque([(n, 0) for n in start_nodes])

        best_dist = None

            try:
                incident_edges = board.node_edges(node)

                try:
                    incident_edges = [e for e in board.edges if node in getattr(board.edges[e], 'nodes', [])]

                return (None, 0.0)

            visited_edges = set()
            visited_nodes = set()
            q = deque([(n, 0) for n in start_nodes])
            visited_nodes.update(start_nodes)

            best_pip = 0.0

                    continue

                    except Exception:

                        best_dist = dist

                # Try multiple ways to get incident edges
                try:
                    incident_edges = None
                    try:
                        incident_edges = board.node_edges(node)
                    except Exception:
                        try:
                            # board.edges may be dict-like keyed by id

                                except Exception:

                            for nn in neigh_nodes:
                                if nn not in visited_nodes:
                                    visited_nodes.add(nn)
                                    q.append((nn, dist + 1))
                        except Exception:
                            continue
                except Exception:
                    pass
            return (best_dist, best_pip)
        except Exception as e:
            print(f"[foo_player] _edge_to_nearest_open_info error: {e}")
            return (None, 0.0)
