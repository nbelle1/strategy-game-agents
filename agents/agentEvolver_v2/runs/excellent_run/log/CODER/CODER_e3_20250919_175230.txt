def decide(self, game, playable_actions):
    # Config
    max_actions = 20
    # Priority order: higher number = higher priority
    PRIORITY = {
        ActionType.BUILD_CITY: 100,
        ActionType.BUILD_SETTLEMENT: 90,
        ActionType.BUILD_ROAD: 80,
        ActionType.BUY_DEV_CARD: 70,
        ActionType.PLAY_DEV_CARD: 60,
        ActionType.TRADE: 50,
        ActionType.MOVE_ROBBER: 40,
        ActionType.END_TURN: 0
    }

    # Helper to get priority (default 10)
    def action_priority(a):
        return PRIORITY.get(a.action_type, 10)

    # 1) Filter END_TURN out if any build/trade/play exists
    has_build_like = any(a.action_type in {
        ActionType.BUILD_CITY,
        ActionType.BUILD_SETTLEMENT,
        ActionType.BUILD_ROAD,
        ActionType.BUY_DEV_CARD,
        ActionType.PLAY_DEV_CARD,
    } for a in playable_actions)

    filtered = []
    for a in playable_actions:
        if a.action_type == ActionType.END_TURN and has_build_like:
            continue
        filtered.append(a)

    if not filtered:
        # nothing but END_TURN -> return it
        return playable_actions[0]  # fallback

    # 2) Cap number of actions to evaluate
    if len(filtered) > max_actions:
        # Keep highest priority actions, then sample the rest deterministically by priority
        filtered = sorted(filtered, key=lambda x: action_priority(x), reverse=True)[:max_actions]

    # 3) Prepare value function
    try:
        value_fn = make_value_fn("base_fn", DEFAULT_WEIGHTS)
    except TypeError:
        # Some environments only require the name
        value_fn = make_value_fn("base_fn")

    best_action = None
    best_score = -float('inf')
    evaluated = 0

    # 4) Simulate each candidate and score resulting state
    for action in filtered:
        try:
            copied = copy_game(game)
            # execute may have different signatures in different forks; try common call
            try:
                execute(copied, action, validate=False)
            except TypeError:
                # fallback to execute(copied, action)
                execute(copied, action)
            # score from the perspective of this player
            score = value_fn(copied, self.color)
        except Exception as e:
            # If simulation fails for any action, skip it and log
            print(f"[foo_player] simulation error for action {action}: {e}")
            continue

        evaluated += 1
        # Use score, then priority as tie-breaker
        if score > best_score or (score == best_score and action_priority(action) > action_priority(best_action)):
            best_score = score
            best_action = action

    # 5) Logging for debugging
    if best_action is None:
        # As a last fallback return first playable action
        print("[foo_player] No simulated action succeeded; falling back to first playable action")
        return playable_actions[0]

    print(f"[foo_player] Chosen action: {best_action}, score: {best_score:.3f}, evaluated: {evaluated}")
    return best_action

def chance_children_for_action(game: Game, action: Action) -> List[Tuple[Game, float]]:
    """
    Mirror AlphaBeta's stochastic modeling:
    - Deterministic actions -> single (next_state, 1.0)
    - BUY_DEVELOPMENT_CARD -> all possible dev cards w/ deck-informed probabilities
    - ROLL -> all dice outcomes 2..12 with number_probability
    - MOVE_ROBBER -> all resource steals (if any) with uniform p=1/5
    """
    return execute_spectrum(game, action)

def chance_children(game: Game, actions: Iterable[Action]) -> Dict[Action, List[Tuple[Game, float]]]:
