def _resolve_edge_nodes(self, board, start_edge):
    """
    Return list of two node ids for an edge or None.
    Tries, in order:
      - board.edge_nodes(start_edge)
      - board.edges[start_edge].nodes
      - if start_edge is tuple/list of length 2 -> list(start_edge)
      - try reversed tuple (start_edge[::-1]) to match key ordering
      - scan board.edges items and match by .nodes equality (order-insensitive)
      - scan board.edges items and match by .id or .index field equal to start_edge
    Logs first N failures to help debugging.
    """
    try:
        # 1: direct board.edge_nodes
        if hasattr(board, 'edge_nodes'):
            try:
                nodes = board.edge_nodes(start_edge)
                if nodes:
                    return list(nodes)
            except Exception:
                pass

        # 2: board.edges[start_edge].nodes
        try:
            edges_attr = getattr(board, 'edges', None)
            if edges_attr is not None and start_edge in edges_attr:
                edge_obj = edges_attr[start_edge]
                nodes = getattr(edge_obj, 'nodes', None)
                if nodes:
                    return list(nodes)
        except Exception:
            pass

        # 3: start_edge is tuple/list
        if isinstance(start_edge, (tuple, list)) and len(start_edge) == 2:
            return list(start_edge)

        # 4: reversed tuple (some boards store edges as reversed keys)
        try:
            if isinstance(start_edge, (tuple, list)) and len(start_edge) == 2:
                rev = (start_edge[1], start_edge[0])
                edges_attr = getattr(board, 'edges', None)
                if edges_attr is not None and rev in edges_attr:

        try:
            edges_attr = getattr(board, 'edges', None)
            if edges_attr:
                for key, edge_obj in getattr(edges_attr, 'items', edges_attr.__iter__ and [])():
                    try:
                        nodes = getattr(edge_obj, 'nodes', None)
                        if nodes:
                            if set(nodes) == set(start_edge if isinstance(start_edge, (list,tuple)) else nodes):
                                return list(nodes)
                    except Exception:
                        pass

def _edge_to_nearest_open_info(self, game, start_edge):
    """Return (distance, pip_sum) for nearest open node reachable from start_edge."""
    try:
        board = game.state.board
        start_nodes = None

        # Try all possible ways to resolve edge -> nodes
        try:
            start_nodes = board.edge_nodes(start_edge)
        except Exception:
            try:
                start_nodes = board.edges[start_edge].nodes
            except Exception:
                try:
                    if isinstance(start_edge, (tuple, list)) and len(start_edge) == 2:
                        start_nodes = list(start_edge)
                    elif hasattr(start_edge, 'nodes'):
                        start_nodes = list(getattr(start_edge, 'nodes', []))
                    elif hasattr(board, 'edges') and start_edge in board.edges:
                        edge_obj = board.edges[start_edge]
                        start_nodes = getattr(edge_obj, 'nodes', None) or getattr(edge_obj, 'nodes', [])
                except Exception:
                    pass

        if not start_nodes:
            print(f"[foo_player] No start_nodes found for edge {start_edge}")
            return (None, 0.0)

        from collections import deque
        visited_edges = set()
        visited_nodes = set()
        q = deque([(n, 0) for n in start_nodes])
        visited_nodes.update(start_nodes)

        best_dist = None
        best_pip = 0.0

        while q:
            node, dist = q.popleft()
            if dist > self.ROAD_MAX_DISTANCE:
                continue
            owner = None
            try:
                owner = board.node_owner(node)
            except Exception:
                try:
                    owner = board.nodes[node].owner
                except Exception:
                    owner = getattr(board.nodes.get(node, {}), 'owner', None)
            if owner is None:
                pip_sum = 0.0
                try:
                    tiles = board.node_tiles(node)
                    for t in tiles:
                        pip = getattr(t, 'pip_probability', getattr(t, 'pip', 1.0))
                        pip_sum += float(pip)
                except Exception:
                    pip_sum = 0.0
                if best_dist is None or dist < best_dist or (dist == best_dist and pip_sum > best_pip):
                    best_dist = dist
                    best_pip = pip_sum
            try:
                incident_edges = board.node_edges(node)
            except Exception:
                incident_edges = [e for e in getattr(board, 'edges', {}) if node in getattr(getattr(board.edges, e, {}), 'nodes', [])]
            for e in incident_edges:
                if e in visited_edges:
                    continue
                visited_edges.add(e)
                try:
                    neigh_nodes = board.edge_nodes(e)
                except Exception:
                    neigh_nodes = getattr(getattr(board.edges, e, {}), 'nodes', [])
                for nn in neigh_nodes:
                    if nn not in visited_nodes:
                        visited_nodes.add(nn)
                        q.append((nn, dist + 1))
        return (best_dist, best_pip)
    except Exception as e:
        print(f"[foo_player] _edge_to_nearest_open_info error: {e}")
        return (None, 0.0)

                    elif hasattr(board, 'edges') and start_edge in board.edges:

            return (None, 0.0)

        visited_edges = set()
        visited_nodes = set()
        q = deque([(n, 0) for n in start_nodes])
        visited_nodes.update(start_nodes)

        best_dist = None
        best_pip = 0.0

        while q:
            node, dist = q.popleft()
            if dist > getattr(self, 'ROAD_MAX_DISTANCE', 6):
                continue

                owner = board.node_owner(node)
            except Exception:

                    owner = board.nodes[node].owner
                except Exception:
                    owner = getattr(board.nodes.get(node, {}), 'owner', None)
            if owner is None:
                pip_sum = 0.0
                try:
                    tiles = board.node_tiles(node)
                    for t in tiles:
                        pip = getattr(t, 'pip_probability', getattr(t, 'pip', 1.0))
                        pip_sum += float(pip)
                except Exception:
                    pip_sum = 0.0
                if best_dist is None or dist < best_dist or (dist == best_dist and pip_sum > best_pip):
                    best_dist = dist
                    best_pip = pip_sum
            try:
                incident_edges = board.node_edges(node)
            except Exception:
                incident_edges = [e for e in getattr(board, 'edges', {}) if node in getattr(getattr(board.edges, e, {}), 'nodes', [])]
            for e in incident_edges:
                if e in visited_edges:
                    continue
                visited_edges.add(e)
                try:
                    neigh_nodes = board.edge_nodes(e)
                except Exception:
                    neigh_nodes = getattr(getattr(board.edges, e, {}), 'nodes', [])
                for nn in neigh_nodes:
                    if nn not in visited_nodes:
                        visited_nodes.add(nn)
                        q.append((nn, dist + 1))
        return (best_dist, best_pip)
    except Exception as e:
        print(f"[foo_player] _edge_to_nearest_open_info error: {e}")
        return (None, 0.0)
